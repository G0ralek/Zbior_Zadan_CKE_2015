funkcja trójkowy(x,k)
    wypisz "0,"
    y = x
    dla i - 1,2,...,k wykonuj
        jeżeli y>=2/3
            wypisz "2"
        jeżeli 2/3>y>=1/3
            wypisz "1"
        jeżeli y<1/3
            wypisz "0"
        y = y*3
        jeżeli y>=2
            y = y - 2
        jeżeli y>=1
            y = y - 1
